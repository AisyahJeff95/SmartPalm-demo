4. EXAMPLE CODE TO CONNECT FRONTEND (JAVASCRIPT FETCH)
------------------------------------------------------------------------
const HF_BASE_URL = "https://aisyjeff95-smartpalm-demo.hf.space";

async function runHuggingFacePrediction(estateName, targetDate, isRealtime = false) {
  const response = await fetch(`${HF_BASE_URL}/api/predict`, {
    method: "POST",
    headers: {
      "Content-Type": "application/json"
    },
    body: JSON.stringify({
      estate_name: estateName,
      date: targetDate,
      is_realtime: isRealtime
    })
  });

  if (!response.ok) {
    throw new Error(`Prediction failed: ${response.status}`);
  }

  const result = await response.json();
  console.log("Prediction Folder:", result.folder_name);
  console.log("Nutrient Overlays:", result.overlays);
  return result;
}
